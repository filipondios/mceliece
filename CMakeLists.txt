cmake_minimum_required(VERSION 3.23)
project(mceliece VERSION 3.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(libsodium REQUIRED)

add_library(mceliece STATIC src/mceliece.cpp)
target_include_directories(mceliece PUBLIC include)
target_link_libraries(mceliece PUBLIC libsodium::libsodium)

include(CTest)
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
